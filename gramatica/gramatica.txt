PROG -> TOPLEVEL PROG | ε
TOPLEVEL -> FUNCDCL | IMPORTDCL | GLOBALDCL
FUNCDCL -> 'fun' 'NAME' '(' PARAMLIST ')' ':' TIPO BLOCO

PARAMLIST -> PARAM MOREPARAMS | ε
MOREPARAMS -> ',' PARAM MOREPARAMS | ε
PARAM -> 'NAME' ':' TIPO

BLOCO -> '{' STMTLIST '}'

STMTLIST -> STMT STMTLIST | ε
STMT -> STMT_SIMPLES | STMT_COMPOSTOS
STMT_SIMPLES -> VARDCL | RETURNSTMT | PRINTSTMT | ATRIBUICAO 
STMT_COMPOSTOS -> WHILESTMT | IFSTMT | DOWHILESTMT | FORSTMT

VARDCL -> 'var' 'NAME' ':' TIPO INICIA ';'
INICIA -> '=' EXPR | ε

ATRIBUICAO -> 'NAME' '=' EXPR ';'

CONDICAO -> ORCOND
ORCOND -> ANDCOND ORCOND_TAIL
ORCOND_TAIL -> '||' ANDCOND ORCOND_TAIL | ε
ANDCOND -> NOTCOND ANDCOND_TAIL
ANDCOND_TAIL -> '&&' NOTCOND ANDCOND_TAIL | ε
NOTCOND -> '!' NOTCOND | EXPR

EXPR ->  TERMO EXPRS
EXPRS -> '+' TERMO EXPRS | '-' TERMO EXPRS | ε
TERMO -> FATOR TERMOS
TERMOS -> '*' FATOR TERMOS | '/' FATOR TERMOS | '%' FATOR TERMOS | ε
FATOR -> '(' EXPR ')' | 'NAME' | 'NAME' '(' ARGLIST ')' |'NUMBER'
ARGLIST -> EXPR MOREARGS | ε
MOREARGS -> ',' EXPR MOREARGS | ε

RETURNSTMT -> 'return' EXPR ';'

TIPO -> 'i32' | 'i64' | 'f32' | 'f64' | 'char'| 'void'

IFSTMT -> 'if' '(' CONDICAO ')' BLOCO ELSESTMT
ELSESTMT -> 'else' BLOCO | 'else' IFSTMT | ε
WHILESTMT -> 'while' '(' CONDICAO ')' BLOCO
DOWHILESTMT -> 'do' BLOCO 'while' '(' CONDICAO ')' ';'
IMPORTDCL -> 'import' 'NAME' ';'
GLOBALDCL -> VARDCL
PRINTSTMT -> 'print' '(' EXPR ')' ';'
FORSTMT -> 'for' '(' VARDCL CONDICAO ')' BLOCO
