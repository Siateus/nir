PROG -> TOPLEVEL PROG | ε

TOPLEVEL -> FUNCDCL | IMPORTDCL | GLOBALDCL

FUNCDCL -> 'fun' 'NAME' '(' PARAMLIST ')' ':' TIPO BLOCO

PARAMLIST -> PARAM MOREPARAMS | ε
MOREPARAMS -> ',' PARAM MOREPARAMS | ε
PARAM -> 'NAME' ':' TIPO

BLOCO -> '{' STMTLIST '}'

STMTLIST -> STMT STMTLIST | ε
STMT -> STMT_SIMPLES | STMT_COMPOSTOS

STMT_SIMPLES -> VARDCL | RETURNSTMT | PRINTSTMT | ATRIBUICAO 

STMT_COMPOSTOS -> FUNCDCL | WHILESTMT | IFSTMT | DOWHILESTMT

VARDCL -> 'NAME' ':' TIPO INICIA ';'
INICIA -> '=' EXPR | ε

ATRIBUICAO -> 'NAME' '=' EXPR ';'

EXPR ->  TERMO EXPRS
EXPRS -> '+' TERMO EXPRS | '-' TERMO EXPRS | ε
TERMO -> FATOR TERMOS
TERMOS -> '*' FATOR TERMOS | '/' FATOR TERMOS | '%' FATOR TERMOS | ε
FATOR -> '(' EXPR ')' | 'NAME' | 'NAME' '(' ARGLIST ')' |'NUMBER'

ARGLIST -> EXPR MOREARGS | ε
MOREARGS -> ',' EXPR MOREARGS | ε

RETURNSTMT -> 'return' EXPR ;

TIPO -> 'INTEGER' | 'FLOAT' | 'CHAR'| 'VOID'

IFSTMT -> 'if' '(' EXPR ')' BLOCO ELSESTMT
ELSESTMT -> 'else' BLOCO | 'else' IFSTMT | ε
